export function toScopeUrl(
  routePath: string,          // basename 뒤 경로. 예: '/indoorshop/zones/assembly'
  search: string,             // 예: '?factory=asm-pbs'
  targetBase: string,         // 예: '/crane_rnd/indoor/'
): string {
  const base = targetBase.endsWith('/') ? targetBase : `${targetBase}/`;
  return `${base}${routePath.replace(/^\//, '')}${search}`;
}

    packages/core/src/config/app-scope.ts (신규)
    export type AppScope = 'crane' | 'indoor';

export interface ResolvedAppScope {
  scope: AppScope;
  /** BrowserRouter basename. 끝 슬래시 없음 ('' | '/crane_rnd' | '/crane_rnd/indoor/stage') */
  basename: string;
}

function withSlash(p: string): string {
  return p.endsWith('/') ? p : `${p}/`;
}

/** 주소만으로 범위를 정한다. indoor 는 끝 슬래시까지 일치해야 한다(/crane_rnd/indoor-work 는 crane). */
export function resolveAppScope(
  pathname: string,
  baseUrl: string,
  indoorBaseUrl: string | undefined,
): ResolvedAppScope {
  const crane = withSlash(baseUrl);
  if (indoorBaseUrl) {
    const indoor = withSlash(indoorBaseUrl);
    if (pathname === indoor.slice(0, -1) || pathname.startsWith(indoor)) {
      return { scope: 'indoor', basename: indoor.slice(0, -1) };
    }
  }
  return { scope: 'crane', basename: crane.slice(0, -1) };
}

    1.4포트: 앱은 33600 에서 3360 · 3362 로

    지금 루트 docker-compose.yml 은 앱을 SHELL_PORT=33600 에 바로 엽니다. 새 구조에서 33600 은 rnd_nginx 의 번호이고, 앱은 그 뒤 3360 + N 으로 내려갑니다.

    33600 이 rnd_nginx 의 번호인 이유

    • 프론트 nginx 의 proxy_pass 와 방화벽이 이미 아는 번호는 33600 하나입니다. 이 번호를 입구로 두면 바깥 설정을 고칠 일이 없습니다.
    • 앱 컨테이너가 33600 을 직접 잡으면 환경이 하나밖에 못 뜹니다. 셋을 띄우려면 앞에서 갈라 줄 nginx 가 그 번호를 가져야 합니다.
    • 환경이 하나 늘어도 바깥은 모릅니다. rnd_nginx location 두 줄과 컨테이너 하나만 늘어납니다.

    앱을 3360 + N 으로 두는 이유

    • N 이 환경: 0 = prod, 1 = stage, 2 = dev. 끝자리만 보면 어느 환경인지 압니다.
    • 33601 · 33602 처럼 입구 바로 옆 번호를 쓰면 "입구가 여럿"으로 읽히고, 33600 대역으로 연 방화벽 규칙에 함께 열릴 수 있습니다. 4자리로 내려 외부 대역과 갈라 둡니다.
    • 3360x 는 바깥에 열지 않습니다. rnd_nginx 가 host.docker.internal:336x 로 붙습니다.
    • indoor 는 별도 포트가 없습니다. 3370 대역은 비워 둡니다.
